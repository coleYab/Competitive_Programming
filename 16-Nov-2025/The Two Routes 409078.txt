struct test {
    void solve() {
        int n, m; 
        cin >> n >> m;
        vector<vector<int>> a(n + 1, vector<int>(n + 1, 0));
        vector<int> d(401);
        for (int i = 1; i <= m; i++) {
            int x, y; cin >> x >> y;
            a[x][y] = a[y][x] = 1;
        }

        if (m == 0 || m == (n * (n + 1)) / 2) {
            cout << -1 << endl;
            return;
        }

        queue<int> q;
        q.push(1);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            if (u == n) {
                cout << d[n] << endl;
                return;
            }

            for (int i = 2; i <= n; i++) {
                if (!d[i] && a[u][i] != a[1][n]) {
                    d[i] = d[u] + 1;
                    q.push(i);
                }
            }
        }
        cout << -1 << endl;
    }
};

int32_t main() {
    ios::sync_with_stdio(false); cin.tie(NULL);
    int tt = 1;
#ifdef multi
    cin >> tt;
#endif
    while (tt--) {
        test().solve();
    }
}
