# Problem: Vowels of All Substrings - https://leetcode.com/problems/vowels-of-all-substrings/


#define ll long long
class Solution {
public:
    bool check(char c) {
        return (c == 'a' || c== 'e' || c == 'i' || c == 'o' || c == 'u');
    }

    long long countVowels(string word) {
        long res = 0, n = word.size();
        for (int i = 0; i < n; ++i)
            if (string("aeiou").find(word[i]) != string::npos)
                res += (i + 1) * (n - i);
        return res;
    }
};
